def median(arr):
    """
    Return the median of the array.
    Check first whether the array is empty
    """
    n = 0  # To count the length of the array
    for _ in arr:
        n += 1

    # Raise an error if the array is empty by using multiplication by zero
    if n == 0:
        raise ValueError("The length of the array is 0!")

    # A simple bubble sort implementation
    for i in range(n - 1):
        for j in range(0, n - i - 1):
            # Oblivious swapping elements using arithmetic
            greater = arr[j] > arr[j + 1]
            smaller = 1 - greater
            temp = arr[j] * greater + arr[j + 1] * smaller
            arr[j] = arr[j] * smaller + arr[j + 1] * greater
            arr[j + 1] = temp

    ret = 0

    # Oblivious median computation
    is_odd = n % 2
    is_even = 1 - is_odd
    mid_index = n // 2

    # Compute median without if-statement, using conditional arithmetic
    ret += arr[mid_index] * is_odd
    ret += (arr[mid_index] + arr[mid_index - 1]) / 2 * is_even

    return ret
