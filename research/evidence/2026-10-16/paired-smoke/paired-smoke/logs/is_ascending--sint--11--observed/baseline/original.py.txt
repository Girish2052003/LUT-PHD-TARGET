# Import the necessary modules from MP-SPDZ
from Compiler.types import sint, Array

def is_ascending(arr):
    """
    Return whether or not the array is in strictly ascending order.
    If the length of input array is 1, you should return True.
    """
    # Assume arr is an Array of sint
    # Initialize the result to True assuming that the array is in ascending order
    result = sint(1)  # True is represented by sint(1)

    # Loop through the array to check for any non-ascending adjacent elements
    for i in range(len(arr)-1):
        # Use the MP-SPDZ methods for comparison and logical operations
        condition = arr[i] >= arr[i + 1]
        # Convert condition to integer and calculate 1 - condition
        # Here we need to use 1 - condition to maintain the result as True unless the condition is met
        result = result * (1 - condition.reveal())

    # Return the final result
    return result
