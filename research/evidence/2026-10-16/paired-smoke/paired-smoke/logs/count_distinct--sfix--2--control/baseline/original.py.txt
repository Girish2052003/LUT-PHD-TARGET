from Compiler.types import sint

def count_distinct(arr):
    """
    Return the number of distinct values in the array using an oblivious approach.
    """
    counter = sint(0)
    n = len(arr)
    for i in range(n):
        unique = sint(1)  # Assume initially every element is unique
        for j in range(i):
            # Update the unique flag obliviously using secure integer operations
            unique *= (arr[j] != arr[i]).reveal()
        
        # Update the counter obliviously using secure integer operations
        counter += unique
    return counter
